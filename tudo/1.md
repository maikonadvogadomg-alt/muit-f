Se possível no ambiente um meio de acesso e consulta ao banco de dados após tudo certo com supabase e mais configurações depois a ser mencionada. a princípio busca o projeto com salvamento no supabase de um editor completo com ia, previus ao vivo  , funcional e que atualizei a cada anexo de arquivos códigos principalmente buscando montar projetos HTML modular. onde também deverá exportar e importar arquivos ou pastas compactadas. a árvore deve manter organizada distinguindo o tipo de código por cor. poder criar pastas e subpastas. estás com poder de nomear, editar, excluir, duplicar renomear, baixar isso para pastas e arquivos. após montagem e teste ao vivo com previus poder baixar e importar projetos que estão pela metade não tendo errado de código e de tipo de linguagem. o segundo é um playground também funcional com salvamento do supabase. o terceiro é um chat de ia profissional com salvamento no supabase de histórico. de chaves, de memória. inicialmente este é um plano e projeto que quero que seja feito de forma completa e com sugestões de melhorias.

    function findMatches(text, searchTerm) {
      if (!searchTerm) return [];
      
      const regex = new RegExp(searchTerm.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'gi');
      const matches = [];
      let match;

      while ((match = regex.exec(text)) !== null) {
        matches.push({
          start: match.index,
          end: match.index + match[0].length,
          text: match[0]
        });
      }

      return matches;
    }

    function highlightMatch(editor, match) {
      if (!match) return;

      const text = editor.value;
      const before = text.substring(0, match.start);
      const lineNum = before.split("\n").length;

      editor.setSelectionRange(match.start, match.end);
      editor.focus();

      const lineHeight = parseInt(window.getComputedStyle(editor).lineHeight);
      editor.scrollTop = (lineNum - 1) * lineHeight - editor.clientHeight / 2;
    }

    function doSearch() {
      const searchTerm = $("searchInput").value;
      const editor = $("codeEditor");

      state.searchMatches = findMatches(editor.value, searchTerm);
      state.searchIndex = 0;

      if (state.searchMatches.length > 0) {
        highlightMatch(editor, state.searchMatches[0]);
        $("searchInfo").textContent = `1/${state.searchMatches.length}`;
      } else {
        $("searchInfo").textContent = "0/0";
      }
    }

    function doReplace() {
      const searchTerm = $("searchInput").value;
      const replaceTerm = $("replaceInput").value;
      const editor = $("codeEditor");

      if (!searchTerm || state.searchMatches.length === 0) return;

      const match = state.searchMatches[state.searchIndex];
      const before = editor.value.substring(0, match.start);
      const after = editor.value.substring(match.end);

      editor.value = before + replaceTerm + after;
      setCurrentCode(editor.value);

      state.searchMatches = findMatches(editor.value, searchTerm);
      if (state.searchMatches.length > 0) {
        highlightMatch(editor, state.searchMatches[0]);
        $("searchInfo").textContent = `1/${state.searchMatches.length}`;
      }
    }

    function doReplaceAll() {
      const searchTerm = $("searchInput").value;
      const replaceTerm = $("replaceInput").value;
      const editor = $("codeEditor");

      if (!searchTerm) return;

      const regex = new RegExp(searchTerm.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'g');
      const newValue = editor.value.replace(regex, replaceTerm);
      const count = (editor.value.match(regex) || []).length;

      editor.value = newValue;
      setCurrentCode(newValue);

      showStatus("success", `${count} ocorrência(s) substituída(s).`);
      doSearch();
    }

    function importFile(file) {
      const reader = new FileReader();

      reader.onload = () => {

        const ext = file.name.split(".").pop().toLowerCase();

          state.html = content;
          state.tab = "html";
        } else if (ext === "css") {
          state.css = content;
          state.tab = "css";
        } else if (ext === "js") {
          state.js = content;
          state.tab = "js";
        } else if (ext === "jsx" || ext === "tsx") {
          state.jsx = content;
          state.tab = "jsx";
        } else {
          state.html = content;
          state.tab = "html";
        }

        persistState();
        renderAll();
        updatePreview();
        showStatus("success", "Arquivo importado: " + file.name);
      };

      reader.readAsText(file);
    }

  function showStatus(type, message, timeout = 5000) {
    const el = $("status");
    el.className = "status show " + type;
    el.textContent = message;

    if (timeout) {
      window.clearTimeout(showStatus.timer);
      showStatus.timer = window.setTimeout(() => {
        el.className = "status";
      }, timeout);
    }
  }

  function loadState() {
    const saved = localStorage.getItem(STORAGE_KEY);
    if (saved) {
      try {
        Object.assign(state, JSON.parse(saved));
      } catch (e) {
        console.error("Erro ao carregar estado:", e);
      }
    }

    const config = localStorage.getItem(CONFIG_KEY);
    if (config) {
      try {
        const cfg = JSON.parse(config);
        $("supabaseUrl").value = cfg.url || "";
        $("supabaseKey").value = cfg.key || "";
      } catch (e) {
        console.error("Erro ao carregar config:", e);
      }
    }

    renderAll();
  }

  function persistState() {
    const copy = {
      tab: state.tab,
      html: state.html,
      css: state.css,
      js: state.js,
      autoRun: state.autoRun,
      editorCollapsed: state.editorCollapsed,
      previewCollapsed: state.previewCollapsed
    };
    localStorage.setItem(STORAGE_KEY, JSON.stringify(copy));
  }

  function getCurrentCode() {
    return state[state.tab] || "";
  }

  function setCurrentCode(value) {
    state[state.tab] = value;
    persistState();
    updateEditor();
    if (state.autoRun) {
      schedulePreview();
    }
  }

  function setTab(tab) {
    state.tab = tab;
    persistState();
    renderAll();
  }

  function updateEditor() {
    const value = getCurrentCode();
    $("codeEditor").value = value;

    const lines = Math.max(1, value.split("\n").length);
    $("lineNumbers").textContent = Array.from(
      { length: lines },
      (_, i) => i + 1
    ).join("\n");

    $("lineCount").textContent = `${lines} ${lines === 1 ? "linha" : "linhas"}`;
  }

  }

    $("editorPanel").classList.toggle("collapsed", state.editorCollapsed);

  let previewTimer = null;

  function updatePreview() {

  }

  function findMatches(text, searchTerm) {
    if (!searchTerm) return [];
    
    const regex = new RegExp(searchTerm.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'gi');
    const matches = [];
    let match;

    while ((match = regex.exec(text)) !== null) {
      matches.push({
        start: match.index,
        end: match.index + match[0].length,
        text: match[0]
      });
    }

    return matches;
  }

  function highlightMatch(editor, match) {
    if (!match) return;

    const text = editor.value;
    const before = text.substring(0, match.start);
    const lineNum = before.split("\n").length;

    editor.setSelectionRange(match.start, match.end);
    editor.focus();

    const lineHeight = parseInt(window.getComputedStyle(editor).lineHeight);
    editor.scrollTop = (lineNum - 1) * lineHeight - editor.clientHeight / 2;
  }

  function doSearch() {
    const searchTerm = $("searchInput").value;
    const editor = $("codeEditor");

    state.searchMatches = findMatches(editor.value, searchTerm);
    state.searchIndex = 0;

    if (state.searchMatches.length > 0) {
      highlightMatch(editor, state.searchMatches[0]);
      $("searchInfo").textContent = `1/${state.searchMatches.length}`;
    } else {
      $("searchInfo").textContent = "0/0";
    }
  }

  function doReplace() {
    const searchTerm = $("searchInput").value;
    const replaceTerm = $("replaceInput").value;
    const editor = $("codeEditor");

    if (!searchTerm || state.searchMatches.length === 0) return;

    const match = state.searchMatches[state.searchIndex];
    const before = editor.value.substring(0, match.start);
    const after = editor.value.substring(match.end);

    editor.value = before + replaceTerm + after;
    setCurrentCode(editor.value);

    state.searchMatches = findMatches(editor.value, searchTerm);
    if (state.searchMatches.length > 0) {
      highlightMatch(editor, state.searchMatches[0]);
      $("searchInfo").textContent = `1/${state.searchMatches.length}`;
    }
  }

  function doReplaceAll() {
    const searchTerm = $("searchInput").value;
    const replaceTerm = $("replaceInput").value;
    const editor = $("codeEditor");

    if (!searchTerm) return;

    const regex = new RegExp(searchTerm.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'g');
    const newValue = editor.value.replace(regex, replaceTerm);
    const count = (editor.value.match(regex) || []).length;

    editor.value = newValue;
    setCurrentCode(newValue);

    showStatus("success", `${count} ocorrência(s) substituída(s).`);
    doSearch();
  }

  async function saveLocal() {
    persistState();
    
    const savedCodes = JSON.parse(localStorage.getItem(SAVED_CODES_KEY) || "[]");
    const newCode = {
      id: Date.now(),
      name: `Código ${new Date().toLocaleTimeString()}`,

      js: state.js,
      date: new Date().toLocaleString()
    };

    showStatus("success", "Salvo localmente.");
  }

  async function saveCloud() {
    if (!state.supabase && !connectSupabase()) {
      showStatus("error", "Configure o Supabase antes.");
      return;
    }

    const payload = {
      name: "Snippet " + new Date().toLocaleTimeString(),
      html: state.html,
      css: state.css,
      js: state.js
    };

    const { error } = await state.supabase
      .from("snippets")
      .insert(payload);

    if (error) {
      showStatus("error", "Erro: " + error.message);
      return;
    }

    state.supabaseConnected = true;
    renderLayout();
    showStatus("success", "Salvo no Supabase.");
  }

  function connectSupabase() {
    const url = $("supabaseUrl").value.trim();
    const key = $("supabaseKey").value.trim();

    if (!url || !key) {
      showStatus("error", "Informe URL e chave.");
      return false;
    }

    if (!window.supabase) {
      showStatus("error", "Biblioteca Supabase não carregada.");
      return false;
    }

    state.supabase = window.supabase.createClient(url, key);
    localStorage.setItem(CONFIG_KEY, JSON.stringify({ url, key }));
    return true;
  }

  function importFile(file) {
    const reader = new FileReader();

    reader.onload = () => {
      const content = String(reader.result || "");
      const ext = file.name.split(".").pop().toLowerCase();

      if (ext === "html" || ext === "htm") {
        state.html = content;
        state.tab = "html";
      } else if (ext === "css") {
        state.css = content;
        state.tab = "css";
      } else if (ext === "js") {
        state.js = content;
        state.tab = "js";
      } else {
        state.html = content;
        state.tab = "html";
      }

      persistState();
      renderAll();
      updatePreview();
      showStatus("success", "Arquivo importado: " + file.name);
    };

    reader.readAsText(file);
  }

    function showStatus(type, message, timeout = 5000) {
      const el = $("status");
      el.className = "status show " + type;
      el.textContent = message;

      if (timeout) {
        window.clearTimeout(showStatus.timer);
        showStatus.timer = window.setTimeout(() => {
          el.className = "status";
        }, timeout);

    function loadState() {
      const saved = localStorage.getItem(STORAGE_KEY);
      if (saved) {
        try {
          Object.assign(state, JSON.parse(saved));
        } catch (e) {
          console.error("Erro ao carregar estado:", e);
        }
      }

      const config = localStorage.getItem(CONFIG_KEY);
      if (config) {
        try {
          const cfg = JSON.parse(config);
          $("supabaseUrl").value = cfg.url || "";
          $("supabaseKey").value = cfg.key || "";
        } catch (e) {
          console.error("Erro ao carregar config:", e);
        }
      }

      renderAll();
    }

    function persistState() {
      const copy = {
        tab: state.tab,
        html: state.html,
        css: state.css,
        js: state.js,
        jsx: state.jsx,
        autoRun: state.autoRun,
        editorCollapsed: state.editorCollapsed,
        previewCollapsed: state.previewCollapsed
      };
      localStorage.setItem(STORAGE_KEY, JSON.stringify(copy));
    }

    function getCurrentCode() {
      return state[state.tab] || "";
    }

    function setCurrentCode(value) {
      state[state.tab] = value;
      persistState();
      updateEditor();
      if (state.autoRun) {
        schedulePreview();
      }
    }

    function setTab(tab) {
      state.tab = tab;
      persistState();
      renderAll();
    }

    function updateEditor() {
      const value = getCurrentCode();
      $("codeEditor").value = value;

      const lines = Math.max(1, value.split("\n").length);
      $("lineNumbers").textContent = Array.from(
        { length: lines },
        (_, i) => i + 1
      ).join("\n");

      $("lineCount").textContent = `${lines} ${lines === 1 ? "linha" : "linhas"}`;
    }

    function renderTabs() {

      });
    }

    function findMatches(text, searchTerm) {
      if (!searchTerm) return [];
      
      const regex = new RegExp(searchTerm.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'gi');
      const matches = [];
      let match;

      while ((match = regex.exec(text)) !== null) {
        matches.push({
          start: match.index,
          end: match.index + match[0].length,
          text: match[0]
        });
      }

      return matches;
    }

    function highlightMatch(editor, match) {
      if (!match) return;

      const text = editor.value;
      const before = text.substring(0, match.start);
      const lineNum = before.split("\n").length;

      editor.setSelectionRange(match.start, match.end);
      editor.focus();

      const lineHeight = parseInt(window.getComputedStyle(editor).lineHeight);
      editor.scrollTop = (lineNum - 1) * lineHeight - editor.clientHeight / 2;
    }

    function doSearch() {
      const searchTerm = $("searchInput").value;
      const editor = $("codeEditor");

      state.searchMatches = findMatches(editor.value, searchTerm);
      state.searchIndex = 0;

      if (state.searchMatches.length > 0) {
        highlightMatch(editor, state.searchMatches[0]);
        $("searchInfo").textContent = `1/${state.searchMatches.length}`;
      } else {
        $("searchInfo").textContent = "0/0";
      }
    }

    function doReplace() {
      const searchTerm = $("searchInput").value;
      const replaceTerm = $("replaceInput").value;
      const editor = $("codeEditor");

      if (!searchTerm || state.searchMatches.length === 0) return;

      const match = state.searchMatches[state.searchIndex];
      const before = editor.value.substring(0, match.start);
      const after = editor.value.substring(match.end);

      editor.value = before + replaceTerm + after;
      setCurrentCode(editor.value);

      state.searchMatches = findMatches(editor.value, searchTerm);
      if (state.searchMatches.length > 0) {
        highlightMatch(editor, state.searchMatches[0]);
        $("searchInfo").textContent = `1/${state.searchMatches.length}`;
      }
    }

    function doReplaceAll() {
      const searchTerm = $("searchInput").value;
      const replaceTerm = $("replaceInput").value;
      const editor = $("codeEditor");

      if (!searchTerm) return;

      const regex = new RegExp(searchTerm.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'g');
      const newValue = editor.value.replace(regex, replaceTerm);
      const count = (editor.value.match(regex) || []).length;

      editor.value = newValue;
      setCurrentCode(newValue);

      showStatus("success", `${count} ocorrência(s) substituída(s).`);
      doSearch();
    }

    function connectSupabase() {

      const key = $("supabaseKey").value.trim();

    function importFile(file) {
      const reader = new FileReader();

      reader.onload = () => {
        const content = String(reader.result || "");
        const ext = file.name.split(".").pop().toLowerCase();

        if (ext === "html" || ext === "htm") {
          state.html = content;
          state.tab = "html";
        } else if (ext === "css") {
          state.css = content;
          state.tab = "css";
        } else if (ext === "js") {
          state.js = content;
          state.tab = "js";
        } else if (ext === "jsx" || ext === "tsx") {
          state.jsx = content;
          state.tab = "jsx";
        } else {
          state.html = content;
          state.tab = "html";
        }

        persistState();
        renderAll();
        updatePreview();
        showStatus("success", "Arquivo importado: " + file.name);
      };

      reader.readAsText(file);
    }

  function loadState() {

  function findMatches(text, searchTerm) {
    if (!searchTerm) return [];
    
    const regex = new RegExp(searchTerm.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'gi');
    const matches = [];
    let match;

    while ((match = regex.exec(text)) !== null) {
      matches.push({
        start: match.index,
        end: match.index + match[0].length,
        text: match[0]
      });
    }

    return matches;
  }

  function highlightMatch(editor, match) {
    if (!match) return;

    const text = editor.value;
    const before = text.substring(0, match.start);
    const lineNum = before.split("\n").length;

    editor.setSelectionRange(match.start, match.end);
    editor.focus();

    const lineHeight = parseInt(window.getComputedStyle(editor).lineHeight);
    editor.scrollTop = (lineNum - 1) * lineHeight - editor.clientHeight / 2;
  }

  function doSearch() {
    const searchTerm = $("searchInput").value;
    const editor = $("codeEditor");

    state.searchMatches = findMatches(editor.value, searchTerm);
    state.searchIndex = 0;

    if (state.searchMatches.length > 0) {
      highlightMatch(editor, state.searchMatches[0]);
      $("searchInfo").textContent = `1/${state.searchMatches.length}`;
    } else {
      $("searchInfo").textContent = "0/0";
    }
  }

  function doReplace() {
    const searchTerm = $("searchInput").value;
    const replaceTerm = $("replaceInput").value;
    const editor = $("codeEditor");

    if (!searchTerm || state.searchMatches.length === 0) return;

    const match = state.searchMatches[state.searchIndex];
    const before = editor.value.substring(0, match.start);
    const after = editor.value.substring(match.end);

    editor.value = before + replaceTerm + after;
    setCurrentCode(editor.value);

    state.searchMatches = findMatches(editor.value, searchTerm);
    if (state.searchMatches.length > 0) {
      highlightMatch(editor, state.searchMatches[0]);
      $("searchInfo").textContent = `1/${state.searchMatches.length}`;
    }
  }

  function doReplaceAll() {
    const searchTerm = $("searchInput").value;
    const replaceTerm = $("replaceInput").value;
    const editor = $("codeEditor");

    if (!searchTerm) return;

    const regex = new RegExp(searchTerm.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'g');
    const newValue = editor.value.replace(regex, replaceTerm);
    const count = (editor.value.match(regex) || []).length;

    editor.value = newValue;
    setCurrentCode(newValue);

    showStatus("success", `${count} ocorrência(s) substituída(s).`);
    doSearch();
  }

  function importFile(file) {
    const reader = new FileReader();

    reader.onload = () => {
      const content = String(reader.result || "");
      const ext = file.name.split(".").pop().toLowerCase();

      if (ext === "html" || ext === "htm") {
        state.html = content;
        state.tab = "html";
      } else if (ext === "css") {
        state.css = content;
        state.tab = "css";
      } else if (ext === "js") {

        state.tab = "js";
      } else {
        state.html = content;
        state.tab = "html";
      }

      persistState();
      renderAll();
      updatePreview();
      showStatus("success", "Arquivo importado: " + file.name);
    };

    reader.readAsText(file);
  }

  function showSavedCodes() {
    const savedCodes = JSON.parse(localStorage.getItem(SAVED_CODES_KEY) || "[]");
    const listEl = $("savedCodesList");

    if (savedCodes.length === 0) {
      listEl.innerHTML = '<p style="color: var(--muted); font-size: 12px;">Nenhum código salvo ainda.</p>';
      $("savedCodesModal").classList.add("open");
      return;
    }

    listEl.innerHTML = savedCodes.map((code, index) => `
      <div class="saved-code-item">
        <div class="saved-code-info">
          <div class="saved-code-name">${code.name}</div>
          <div class="saved-code-date">${code.date}</div>
        </div>
        <div class="saved-code-actions">
          <button class="btn primary" onclick="loadSavedCode(${index})">Carregar</button>
          <button class="btn danger" onclick="deleteSavedCode(${index})">Deletar</button>
        </div>
      </div>
    `).join("");

    $("savedCodesModal").classList.add("open");
  }

  function loadSavedCode(index) {
    const savedCodes = JSON.parse(localStorage.getItem(SAVED_CODES_KEY) || "[]");
    const code = savedCodes[index];

    if (!code) {
      showStatus("error", "Código não encontrado.");
      return;
    }

    state.html = code.html;
    state.css = code.css;
    state.js = code.js;
    state.tab = "html";

    persistState();
    renderAll();
    updatePreview();

    $("savedCodesModal").classList.remove("open");
    showStatus("success", `Código carregado: ${code.name}`);
  }

  function deleteSavedCode(index) {
    if (!confirm("Tem certeza que deseja deletar este código?")) return;

    const savedCodes = JSON.parse(localStorage.getItem(SAVED_CODES_KEY) || "[]");
    savedCodes.splice(index, 1);
    localStorage.setItem(SAVED_CODES_KEY, JSON.stringify(savedCodes));

    showStatus("success", "Código deletado.");
    showSavedCodes();
  }
</script>
